cmake_minimum_required(VERSION 3.20)
project(MoonRiverStudio VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(mrs_core STATIC
  core/src/model.cpp core/src/timeline.cpp core/src/transport.cpp
  core/src/project_store.cpp core/src/serialization.cpp core/src/fixtures.cpp core/src/musical.cpp core/src/arrangement.cpp)
target_include_directories(mrs_core PUBLIC core/include)
target_compile_features(mrs_core PUBLIC cxx_std_20)
set_target_properties(mrs_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(mrs_core PUBLIC /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(mrs_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(mrs_core_check tools/core_check.cpp)
target_link_libraries(mrs_core_check PRIVATE mrs_core)
if(BUILD_TESTING)
  add_executable(mrs_core_tests tests/core_tests.cpp)
  target_link_libraries(mrs_core_tests PRIVATE mrs_core)
  foreach(suite model timeline transport commands serialization integration)
    add_test(NAME core_${suite} COMMAND mrs_core_tests ${suite})
  endforeach()
  add_test(NAME core_acceptance COMMAND mrs_core_check)
endif()

add_library(mrs_processing STATIC
  core/processing/src/model.cpp core/processing/src/native.cpp core/processing/src/graph.cpp)
target_link_libraries(mrs_processing PUBLIC mrs_core)
target_include_directories(mrs_processing PRIVATE core/audio/include)

find_package(Threads REQUIRED)
add_library(mrs_audio STATIC
  core/audio/src/engine.cpp core/audio/src/wav.cpp core/audio/src/device.cpp core/audio/src/offline_device.cpp core/audio/src/waveform.cpp core/audio/src/read_ahead.cpp core/audio/src/recording.cpp)
target_include_directories(mrs_audio PUBLIC core/audio/include)
target_link_libraries(mrs_audio PUBLIC mrs_core Threads::Threads mrs_processing)
add_executable(mrs_audio_check tools/audio_check.cpp)
target_link_libraries(mrs_audio_check PRIVATE mrs_audio)
if(BUILD_TESTING)
  add_executable(mrs_audio_tests tests/audio_tests.cpp)
  target_link_libraries(mrs_audio_tests PRIVATE mrs_audio)
  foreach(suite render transport queue wav device metrics independence streaming recording)
    add_test(NAME audio_${suite} COMMAND mrs_audio_tests ${suite})
  endforeach()
  add_test(NAME audio_acceptance COMMAND mrs_audio_check --self-test)
endif()
option(MRS_BUILD_ASIO "Build the Windows vendor ASIO device prototype" OFF)
if(MRS_BUILD_ASIO)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Stage 1 ASIO prototype requires Windows x64")
  endif()
  include(cmake/PortAudio.cmake)
  add_library(mrs_asio STATIC core/audio/src/portaudio_asio.cpp)
  target_link_libraries(mrs_asio PUBLIC mrs_audio PRIVATE portaudio)
  target_link_libraries(mrs_audio_check PRIVATE mrs_asio)
  target_compile_definitions(mrs_audio_check PRIVATE MRS_HAS_ASIO=1)
endif()

add_executable(mrs_timeline_check tools/timeline_check.cpp)
target_link_libraries(mrs_timeline_check PRIVATE mrs_audio)
if(BUILD_TESTING)
  add_executable(mrs_musical_tests tests/musical_tests.cpp)
  target_link_libraries(mrs_musical_tests PRIVATE mrs_audio)
  foreach(suite model context navigation edits persistence engine)
    add_test(NAME musical_${suite} COMMAND mrs_musical_tests ${suite})
  endforeach()
  add_test(NAME musical_acceptance COMMAND mrs_timeline_check)
endif()

add_executable(mrs_processor_check tools/processor_check.cpp)
target_link_libraries(mrs_processor_check PRIVATE mrs_audio)
if(BUILD_TESTING)
  add_executable(mrs_processing_tests tests/processing_tests.cpp)
  target_link_libraries(mrs_processing_tests PRIVATE mrs_audio)
  foreach(suite midi model graph parameters state latency engine realtime)
    add_test(NAME processing_${suite} COMMAND mrs_processing_tests ${suite})
  endforeach()
  add_test(NAME processing_acceptance COMMAND mrs_processor_check)
endif()

add_library(mrs_persistence STATIC core/persistence/src/archive.cpp core/persistence/src/files.cpp)
target_link_libraries(mrs_persistence PUBLIC mrs_processing Threads::Threads)
add_executable(mrs_persistence_check tools/persistence_check.cpp)
target_link_libraries(mrs_persistence_check PRIVATE mrs_persistence)
if(BUILD_TESTING)
  add_executable(mrs_persistence_tests tests/persistence_tests.cpp)
  target_link_libraries(mrs_persistence_tests PRIVATE mrs_persistence)
  foreach(suite model roundtrip migrations compatibility show atomic recovery autosave session)
    add_test(NAME persistence_${suite} COMMAND mrs_persistence_tests ${suite})
  endforeach()
  add_test(NAME persistence_acceptance COMMAND mrs_persistence_check)
endif()

add_library(mrs_desktop STATIC apps/studio-desktop/src/application.cpp apps/studio-desktop/src/project_folders.cpp)
target_include_directories(mrs_desktop PUBLIC apps/studio-desktop/include)
target_link_libraries(mrs_desktop PUBLIC mrs_audio mrs_persistence)
if(BUILD_TESTING)
  add_executable(mrs_desktop_tests tests/desktop_tests.cpp)
  target_link_libraries(mrs_desktop_tests PRIVATE mrs_desktop)
  foreach(suite workspaces transport files assets audio config arrangement waveform nonplaying_edits clip_edits streaming mono_route recording project_folders)
    add_test(NAME desktop_${suite} COMMAND mrs_desktop_tests ${suite})
  endforeach()
endif()
option(MRS_BUILD_DESKTOP "Build the native Windows DAW shell" ON)
if(WIN32 AND MRS_BUILD_DESKTOP)
  enable_language(RC)
  add_executable(MoonRiverStudio WIN32 apps/studio-desktop/src/win32_main.cpp apps/studio-desktop/studio.rc)
  target_include_directories(MoonRiverStudio PRIVATE apps/studio-desktop)
  target_compile_definitions(MoonRiverStudio PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(MoonRiverStudio PRIVATE mrs_desktop user32 gdi32 comdlg32 shell32 ole32 uuid)
  if(MSVC)
    target_link_options(MoonRiverStudio PRIVATE /MANIFEST:NO)
  endif()
  if(MRS_BUILD_ASIO)
    target_link_libraries(MoonRiverStudio PRIVATE mrs_asio)
    target_compile_definitions(MoonRiverStudio PRIVATE MRS_HAS_ASIO=1)
  elseif(BUILD_TESTING)
    add_test(NAME desktop_gui_smoke COMMAND MoonRiverStudio --smoke-test)
    set_tests_properties(desktop_gui_smoke PROPERTIES TIMEOUT 20)
  endif()
endif()
